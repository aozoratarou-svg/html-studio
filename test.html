// --- 動的ステータス更新用ヘルパー（既存のものを拡張） --- function updateStatus(agentId, state, text) { const el = document.getElementById(agentId); if (el) { el.className = `agent-status status-${state}`; el.innerText = text; } } // --- 新しい自律型マルチエージェント・ワークフロー --- async function runMultiAgent() { const input = document.getElementById('userInput').value; const key = localStorage.getItem('gemini_key'); if(!key) { alert('⚙️ カラ Gemini API Key ヲ ニュウリョクシテクダサイ。'); return; } if(!input) { alert('ヨウボウ ヲ ニュウリョクシテクダサイ。'); return; } const outputArea = document.getElementById('output'); const statusBoard = document.getElementById('status-board'); const startBtn = document.getElementById('startBtn'); // UI初期化（固定の子エージェントではなく、動的に生成するため一旦クリア） outputArea.value = ""; statusBoard.style.display = "block"; statusBoard.innerHTML = '
🧠 親(PM) : チーム編成中... (Function Calling)
'; startBtn.disabled = true; startBtn.innerText = "⏳ ギョウムチュウ..."; try { // 【ステップ1】PMに持たせる「部下召喚ツール」の定義（JSON Schema） const spawnTool = { functionDeclarations: [{ name: "spawn_agents", description: "ユーザーの要望を達成するために必要な専門エージェントのチームを編成します。", parameters: { type: "OBJECT", properties: { agents: { type: "ARRAY", description: "必要な子エージェントのリスト", items: { type: "OBJECT", properties: { role: { type: "STRING", description: "役割 (例: 'UI Engineer', 'Backend', 'Translator', 'Reviewer')" }, task: { type: "STRING", description: "この子エージェントが実行すべき具体的なタスク指示" } } } } }, required: ["agents"] } }] }; // 【ステップ2】PMエージェントを呼び出し、ツールを使わせる const pmPrompt = `あなたは優秀なPMです。以下の要望を分析し、タスクを分解して 'spawn_agents' ツールを実行してください。必要な人数はあなたが自由に決めて構いません。\n\n要望: ${input}`; const pmRes = await fetch(`${API_URL}?key=${key}`, { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ contents: [{ parts: [{ text: pmPrompt }] }], tools: [spawnTool], // ★ここでPMにツールを渡す generationConfig: { temperature: 0.2 } }) }); if (!pmRes.ok) throw new Error(`API Error: ${pmRes.status}`); const pmData = await pmRes.json(); // 【ステップ3】PMの判断（ツールの実行要求）を解析する const parts = pmData.candidates[0].content.parts; const functionCall = parts.find(p => p.functionCall)?.functionCall; if (!functionCall || functionCall.name !== "spawn_agents") { throw new Error("PMがツールを使用せず、テキストを返しました。やり直してください。"); } const agentTeam = functionCall.args.agents; document.getElementById('status-pm').className = "agent-status status-done"; document.getElementById('status-pm').innerText = `🧠 親(PM) : 編成完了！ ${agentTeam.length}名のエージェントを召喚します。`; // 【ステップ4】編成された人数の分だけ、動的にUIへステータスを追加 agentTeam.forEach((agent, index) => { statusBoard.innerHTML += `
🤖 子[${agent.role}]: 待機中...
`; }); // 【ステップ5】子エージェントたちの並列実行（Swarm） const childPromises = agentTeam.map(async (agent, index) => { updateStatus(`status-child-${index}`, 'run', `🤖 子[${agent.role}]: タスク実行中...`); // 子エージェント用のプロンプト const childPrompt = `あなたは専門家「${agent.role}」です。以下のタスクのみを実行し、挨拶や解説を省いて純粋な結果のみを出力してください。\n\n【タスク】:\n${agent.task}`; // （※必要に応じて、ここで先ほどのrunCodeAgentWithRetryを使うことも可能です） const result = await callGemini(childPrompt, key, 0.0); updateStatus(`status-child-${index}`, 'done', `🤖 子[${agent.role}]: 完了！`); // 出力フォーマットを整える return `### 担当: ${agent.role}\n${result}\n\n`; }); // 全員が完了するのを待つ const results = await Promise.all(childPromises); // 【ステップ6】最終統合出力 outputArea.value = "=== 8-BIT SWARM OUTPUT ===\n\n" + results.join(""); } catch(e) { outputArea.value = "ERROR: " + e.message; console.error(e); updateStatus('status-pm', 'err', '🧠 親(PM) : エラー停止'); } finally { startBtn.disabled = false; startBtn.innerText = "🕹️ START (生成)"; } }
